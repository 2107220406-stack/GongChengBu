CREATE DATABASE IF NOT EXISTS word_lab CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE TABLE IF NOT EXISTS analysis_records (id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY, text TEXT NOT NULL, score DECIMAL(3,2) NOT NULL, label VARCHAR(16) NOT NULL, pinyin TEXT NOT NULL, created_at DATETIME NOT NULL, created_at_ts TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
